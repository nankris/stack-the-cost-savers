Bug report: some URLs still point at the legacy host. The bug is in `T10_replace_all/broken.js`. Check with `cd T10_replace_all && node test.js` (it should print OK / exit 0). Do not edit any test.js and don't change other packages.

Current `T10_replace_all/broken.js`:
```js
const API = "http://legacy.example";

function url(path) {
  return "http://legacy.example" + path;
}

function health() {
  return "http://legacy.example" + "/health";
}

module.exports = { API, url, health };

```
`T10_replace_all/test.js`:
```js
const assert = require("assert");
const { API, url, health } = require("./broken");
assert.strictEqual(API, "http://api.example");
assert.strictEqual(url("/v1"), "http://api.example/v1");
assert.strictEqual(health(), "http://api.example/health");
console.log("OK");

```
